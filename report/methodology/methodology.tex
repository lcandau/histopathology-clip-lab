\section{Histopathology Digital Imaging}
\label{sec:meth-histo}
\begin{itemize}
    \item H\&E staining.
    \item Whole-Slide Imaging + patch extraction.
    \item Stain variability and normalization (Macenko, Reinhard, Vahadane).
    \item Why the domain is hard for general-purpose VLMs: domain gap from natural images, no per-image captions in clinical archives, dataset artifacts that can drive performance.
\end{itemize}


\section{Contrastive Vision-Language Learning (CLIP)}
\label{sec:meth-clip}

\subsection{Why CLIP?}
\begin{itemize}
    \item CLIP as object of study, not the classifier.
    \item Capabilities CLIP provides over a softmax head:
    \begin{itemize}
        \item Open-vocabulary classification (new classes via prompts proven by adding new samples).
        \item Semantic similarity output (cosine vs natural language).
        \item Prompt engineering as a sensitivity lever.
    \end{itemize}
\end{itemize}

\subsection{Architecture and training signal}
\begin{itemize}
    \item Dual-encoder + shared embedding space.
    \item InfoNCE loss, learnable temperature.
    \item In-batch false-negative regime at small batch + few classes.
    \item Prompt-based zero-shot classification.
    \item CLIP-paper sample-level retrieval $\to$ doesn't apply (class-shared prompts).
\end{itemize}

\section{Reference baselines: PLIP and a plain image classifier}
\label{sec:meth-reference}
\begin{itemize}
    \item PLIP zero-shot (Huang 2023) --- pathology-pretrained CLIP, frozen, no LC25000 training.
    \item Plain image classifier --- Dense(5)+softmax on frozen ResNet50 trained with cross-entropy; no text encoder, no CLIP architecture.
    \item Together: PLIP isolates the pretraining-domain contribution; the plain classifier isolates the CLIP-architecture contribution.
\end{itemize}


\section{Dataset: Kaggle LC25000}
\label{sec:meth-dataset}
\begin{itemize}
    \item Borkowski 2019.
    \item 5 tissue classes (lung, colon).
    \item 25\,000 patches from 1\,250 originals via rotation/flip $\to$ no further geometric augmentation.
    \item Known limits: single institution, possible stain artifacts, no patient-level grouping.

\section{Preprocessing and Input Pipeline}
\label{sec:meth-pre}
\begin{itemize}
    \item Image: resize $224\times224$, ResNet-style preprocessing, no further augmentation.
    \item Text: BERT WordPiece, fixed sequence length.
    \item Split: stratified 80/10/10, seed 42, persisted.
\end{itemize}


\section{The Baseline}
\label{sec:meth-baseline}
\begin{itemize}
    \item Frozen ResNet50 + frozen BERT.
    \item Projection heads (Dense + LayerNorm) $\to$ 256-d shared space.
    \item Learnable temperature, CLIP-style init.
    \item Symmetric InfoNCE, AdamW, grad clipping.
    \item Single fixed prompt at train + eval.
    \item Float32 throughout.
\end{itemize}

\section{Evaluation Protocol}
\label{sec:meth-eval}
\begin{itemize}
    \item Classification: argmax cosine(image, class-prompt).
    \item Global: accuracy, balanced accuracy, macro-F1, weighted-F1.
    \item Per-class: precision, recall, F1.
    \item Confusion matrices: raw + row-normalized.
    \item Embeddings: shared image+prompt UMAP, centroid-to-prompt heatmap.
    \item Class-level retrieval, with explicit positive definition.
\end{itemize}


\section{Experimental Design and Ablation Protocol}
\label{sec:meth-design}
\begin{itemize}
    \item Single-variable-change principle.
    \item Matrix grouped by axis of variation; full matrix in §\ref{sec:experiment-matrix}.
    \item Scope boundaries: no patient-level grouping, no human-expert evaluation, no architecture-controlled comparison vs CONCH / BiomedCLIP / MUSK. Deferred to future work.
\end{itemize}
